### 2.2.2 Canvas Sizing

**Problem:** Canvas doesn't respect selected output ratio.

**Fix:** Bind canvas dimensions to format selector:

```javascript
function setCanvasFormat(ratio) {
  const canvas = document.getElementById('stage-canvas');
  const container = document.getElementById('canvas-container');
  
  const formats = {
    '9:16': { w: 1080, h: 1920 },
    '1:1': { w: 1080, h: 1080 },
    '16:9': { w: 1920, h: 1080 }
  };
  
  const { w, h } = formats[ratio];
  canvas.width = w;
  canvas.height = h;
  
  // Scale container to fit viewport while maintaining aspect
  const containerMaxW = container.parentElement.clientWidth - 48;
  const scale = Math.min(containerMaxW / w, (window.innerHeight - 200) / h);
  container.style.width = `${w * scale}px`;
  container.style.height = `${h * scale}px`;
}
```

**Required CSS Variables:**

  --bg-base: #0a0d12;
  --bg-elevated: #11141a;

  --text-primary: #f0f2f5;
  --text-secondary: #a0a8b8;
  --text-muted: #6b7280;

  --accent-secondary: #8b5cf6;

  --radius-md: 10px;
  --radius-lg: 14px;

**Apply to ALL elements:**

```css
* {
  margin: 0;
  padding: 0;
  box-sizing: border-box;
}

  color: var(--text-primary);

  border-radius: var(--radius-md);
}

input:focus, select:focus, textarea:focus {
  outline: none;
  border-color: var(--accent-primary);
}
```
