## A.3 State Management

```javascript
// Simple event-driven state (no Redux needed)
const store = {
  state: {},
  
  set(key, value) {
    this.state[key] = value;
    this.emit(key, value);
    localStorage.setItem(`swr-${key}`, JSON.stringify(value));
  },
  
  get(key) {
    return this.state[key] ?? JSON.parse(localStorage.getItem(`swr-${key}`));
  },
  
  listeners: {},
  
  on(key, fn) {
    this.listeners[key] = this.listeners[key] || [];
    this.listeners[key].push(fn);
  },
  
  emit(key, value) {
    (this.listeners[key] || []).forEach(fn => fn(value));
  }
};
```
