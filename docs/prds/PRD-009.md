# PRD-009: CLIENT REVIEW — Phase 7

## 9.1 Overview

No backend, but clients need to review and comment.

## 9.2 Requirements

### 9.2.1 Watermarked Preview

Low-bitrate export: 1 Mbps, 720p. Burn "PREVIEW — NOT FOR DISTRIBUTION" at 25% opacity center, or subtle SWR logo bug.

### 9.2.2 Self-Contained Review Page

Generate standalone HTML file embedding watermarked MP4:

```javascript
function generateReviewPage(videoBlob, metadata) {
  const reader = new FileReader();
  reader.onload = () => {
    const html = `<!DOCTYPE html>
<html>
<head>
  <title>SWR Review — ${metadata.song}</title>
  <style>
    body { margin:0; background:#0a0d12; color:#fff; font-family:sans-serif; }
    video { width:100%; max-width:960px; display:block; margin:0 auto; }
    .meta { padding:20px; text-align:center; color:#888; }
    .pins { position:relative; }
    .pin { position:absolute; width:20px; height:20px; background:#6366f1; 
           border-radius:50%; cursor:pointer; }
    .comment { display:none; position:absolute; background:#1a1e28; padding:10px; 
               border-radius:8px; width:200px; }
    .pin:hover .comment { display:block; }
  </style>
</head>
<body>
  <div class="meta">
    <h2>${metadata.song}</h2>
    <p>${metadata.date} · ${metadata.preset}</p>
  </div>
  <div class="pins">
    <video src="${reader.result}" controls></video>
  </div>
  <script>
    // Pin drop logic
    document.querySelector('video').addEventListener('click', (e) => {
      const rect = e.target.getBoundingClientRect();
      const x = ((e.clientX - rect.left) / rect.width * 100).toFixed(1);
      const y = ((e.clientY - rect.top) / rect.height * 100).toFixed(1);
      const note = prompt('Comment:');
      if (note) addPin(x, y, note);
    });
    function addPin(x, y, note) {
      const pin = document.createElement('div');
      pin.className = 'pin';
      pin.style.left = x + '%';
      pin.style.top = y + '%';
      pin.innerHTML = '<div class="comment">' + note + '</div>';
      document.querySelector('.pins').appendChild(pin);
    }
  </script>
</body>
</html>`;
    downloadAsFile(html, `review-${metadata.song}.html`);
  };
  reader.readAsDataURL(videoBlob);
}
```
